API

Use the StreamX API in your own site

Free JSON. No API key. Two requests are all you need.

1. Get the category list

Call this once when your page or app starts.

GET https://zk5740764-cmyk.github.io/Streamx-/categories/manifest.json

2. Get the videos of a category

GET https://zk5740764-cmyk.github.io/Streamx-/categories/<name>.json

Available names: home recently_uploaded indian_pop_music haryanvi punjabi storytelling music gaming news sports fashion bollywood_movies hollywood_movies south_indian_movies punjabi_movies

Video fields

FieldMeaning
titleVideo title
urlYouTube link; the video ID is read from it
thumbnailPreview image
channelChannel name
durationVideo length

How to use it

  1. Fetch manifest.json and show the names as tabs or a menu.
  2. When a category is picked, fetch its .json file and draw the video cards.
  3. On a website, play with the YouTube embed on an HTTPS page. In an Android app, play with ExoPlayer.
  4. To update data, edit the JSON files in your GitHub repo and push. GitHub Pages updates in a few minutes.

<!DOCTYPE html>
<html>
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">

</head>

const BASE = "https://zk5740764-cmyk.github.io/Streamx-/categories/";
const $ = id => document.getElementById(id);

async function start() {
  const m = await (await fetch(BASE + "manifest.json")).json();
  const cats = m.categories || m.list || [];
  $("cat").innerHTML = cats.map(c => {
    const k = c.file || c.name || c.id;
    return `<option value="${k}">${k}</option>`;
  }).join("");
  $("cat").onchange = () => loadCategory($("cat").value);
  loadCategory($("cat").value);
}

async function loadCategory(key) {
  const file = key.endsWith(".json") ? key : key + ".json";
  const data = await (await fetch(BASE + file)).json();
  const videos = Array.isArray(data) ? data : (data.videos || data.items || []);
  $("list").innerHTML = "";
  videos.forEach(v => {
    const d = document.createElement("div");
    d.innerHTML = `<img src="${v.thumbnail}" width="160"> ${v.title}`;
    d.onclick = () => play(v);
    $("list").appendChild(d);
  });
}

function play(v) {
  const u = new URL(v.url);
  const id = u.hostname.includes("youtu.be") ? u.pathname.slice(1) : u.searchParams.get("v");
  $("player").innerHTML = `<iframe width="100%" height="315"
    src="https://www.youtube.com/embed/${id}?autoplay=1"
    referrerpolicy="strict-origin-when-cross-origin" allowfullscreen></iframe>`;
}
start();
</script>
</body>
</html>

  static Future<List> getCategories() async {
    final res = await http.get(Uri.parse("$BASE/manifest.json"));
    final data = jsonDecode(res.body);
    return data["categories"];

    final res = await http.get(Uri.parse("$BASE/$fileName"));

package com.example.streamx

import android.content.Intent
import android.net.Uri
import android.os.Bundle
import android.widget.ArrayAdapter
import android.widget.ListView
import androidx.appcompat.app.AppCompatActivity
import kotlinx.coroutines.*
import org.json.JSONArray
import org.json.JSONObject
import java.net.URL

class MainActivity : AppCompatActivity() {

    private val BASE = "https://zk5740764-cmyk.github.io/Streamx-/categories"
    private lateinit var categoriesList: ListView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        categoriesList = findViewById(R.id.categoriesList)
        loadCategories()
    }

    private fun loadCategories() {
        CoroutineScope(Dispatchers.IO).launch {
            try {
                val json = URL("$BASE/manifest.json").readText()
                val data = JSONObject(json)
                val categories = data.getJSONArray("categories")

                val names = ArrayList<String>()
                for (i in 0 until categories.length()) {
                    val cat = categories.getJSONObject(i)
                    names.add("${cat.getString("name")} (${cat.getInt("count")})")
                }

                runOnUiThread {
                    categoriesList.adapter = ArrayAdapter(
                        this@MainActivity,
                        android.R.layout.simple_list_item_1,
                        names
                    )
                    categoriesList.setOnItemClickListener { _, _, pos, _ ->
                        val cat = categories.getJSONObject(pos)
                        loadVideos(cat.getString("file"))
                    }
                }
            } catch (e: Exception) {
                e.printStackTrace()
            }
        }
    }

    private fun loadVideos(fileName: String) {
        CoroutineScope(Dispatchers.IO).launch {
            try {
                val json = URL("$BASE/$fileName").readText()
                val videos = JSONArray(json)

                val titles = ArrayList<String>()
                val urls = ArrayList<String>()
                for (i in 0 until videos.length()) {
                    val v = videos.getJSONObject(i)
                    titles.add(v.getString("title"))
                    urls.add(v.getString("url"))
                }

                runOnUiThread {
                    categoriesList.adapter = ArrayAdapter(
                        this@MainActivity,
                        android.R.layout.simple_list_item_1,
                        titles
                    )
                    categoriesList.setOnItemClickListener { _, _, pos, _ ->
                        playVideo(urls[pos])
                    }
                }
            } catch (e: Exception) {
                e.printStackTrace()
            }
        }
    }

    private fun playVideo(url: String) {
        startActivity(Intent(this, PlayerActivity::class.java).putExtra("url", url))
    }
}

Why Error 153 happens

The YouTube embed refuses to start when the page sends no referrer: the HTML is opened as a local file, or an Android WebView loads it with loadData(). Hosting on HTTPS fixes the web demo. In the app, skip the embed and play with ExoPlayer.

How ExoPlayer plays YouTube

ExoPlayer cannot open a youtube.com/watch link. It needs a direct stream URL, so NewPipeExtractor converts the link first.

1
Add dependencies

Media3 ExoPlayer, NewPipeExtractor, OkHttp, desugaring

2
Resolve the url field

StreamInfo.getInfo() returns video and audio stream URLs

3
Play

Merge video and audio in ExoPlayer, show it in PlayerView

NewPipeDownloader.kt

NewPipeDownloader.kt
import okhttp3.OkHttpClient
import okhttp3.RequestBody.Companion.toRequestBody
import org.schabi.newpipe.extractor.downloader.*

class NewPipeDownloader : Downloader() {
    private val client = OkHttpClient()
    override fun execute(request: Request): Response {
        val body = request.dataToSend()?.toRequestBody()
        val b = okhttp3.Request.Builder().url(request.url()).method(request.httpMethod(), body)
        request.headers().forEach { (k, v) -> b.removeHeader(k); v.forEach { b.addHeader(k, it) } }
        client.newCall(b.build()).execute().use { r ->
            return Response(r.code, r.message, r.headers.toMultimap(), r.body?.string(), r.request.url.toString())
        }
    }
}

PlayerActivity.kt

PlayerActivity.kt
import android.os.Bundle
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import androidx.media3.common.MediaItem
import androidx.media3.datasource.DefaultHttpDataSource
import androidx.media3.exoplayer.ExoPlayer
import androidx.media3.exoplayer.source.MergingMediaSource
import androidx.media3.exoplayer.source.ProgressiveMediaSource
import androidx.media3.ui.PlayerView
import kotlinx.coroutines.*
import org.schabi.newpipe.extractor.NewPipe
import org.schabi.newpipe.extractor.ServiceList
import org.schabi.newpipe.extractor.stream.StreamInfo

class PlayerActivity : AppCompatActivity() {
    private var player: ExoPlayer? = null

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        val view = PlayerView(this)
        setContentView(view)
        val url = intent.getStringExtra("url") ?: return finish()

        lifecycleScope.launch {
            try {
                // 1) Resolve the YouTube link into direct stream URLs (off the main thread)
                val (video, audio) = withContext(Dispatchers.IO) {
                    NewPipe.init(NewPipeDownloader())
                    val info = StreamInfo.getInfo(ServiceList.YouTube, url)
                    // Best HD: video-only + audio, merged. Simple option: info.videoStreams (up to 720p, has audio)
                    val v = info.videoOnlyStreams.maxByOrNull { it.height }
                    val a = info.audioStreams.maxByOrNull { it.averageBitrate }
                    if (v != null && a != null) v.content to a.content
                    else info.videoStreams.maxByOrNull { it.height }!!.content to null
                }
                // 2) Build the ExoPlayer media source
                val factory = DefaultHttpDataSource.Factory().setUserAgent("Mozilla/5.0")
                val vSrc = ProgressiveMediaSource.Factory(factory).createMediaSource(MediaItem.fromUri(video))
                val source = if (audio != null)
                    MergingMediaSource(vSrc, ProgressiveMediaSource.Factory(factory).createMediaSource(MediaItem.fromUri(audio)))
                else vSrc
                // 3) Play
                player = ExoPlayer.Builder(this@PlayerActivity).build().also {
                    view.player = it
                    it.setMediaSource(source)
                    it.prepare()
                    it.playWhenReady = true
                }
            } catch (e: Exception) {
                Toast.makeText(this@PlayerActivity, "Could not play: ${e.message}", Toast.LENGTH_LONG).show()
            }
        }
    }

    override fun onDestroy() { player?.release(); super.onDestroy() }
}

Play from this web page with ExoPlayer

A browser cannot run ExoPlayer. But when this page is opened inside your app's WebView, tapping a video calls the app and ExoPlayer plays it. In a normal browser the page falls back to the YouTube embed.

WebActivity.kt
import android.content.Intent
import android.os.Bundle
import android.webkit.JavascriptInterface
import android.webkit.WebView
import androidx.appcompat.app.AppCompatActivity

class WebActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        val web = WebView(this)
        setContentView(web)
        web.settings.javaScriptEnabled = true
        web.settings.domStorageEnabled = true
        // The page calls AndroidPlayer.play(url) -> ExoPlayer opens
        web.addJavascriptInterface(object {
            @JavascriptInterface
            fun play(url: String) {
                startActivity(Intent(this@WebActivity, PlayerActivity::class.java).putExtra("url", url))
            }
        }, "AndroidPlayer")
        web.loadUrl("https://YOUR-USERNAME.github.io/StreamX_API_v2.html")
    }
}

  if (currentFile) {
    return (

          <Text style={{ fontSize: 18, marginBottom: 10 }}>&larr; Back</Text>

          renderItem={({ item }) => (
            <TouchableOpacity
              onPress={() => Linking.openURL(item.url)}
              style={styles.video}
            >
              <Image source={{ uri: item.thumbnail }} style={styles.thumb} />
              <View style={{ flex: 1 }}>
                <Text numberOfLines={2}>{item.title}</Text>
                <Text style={{ color: '#888' }}>{item.channel}</Text>
              </View>
            </TouchableOpacity>
          )}
        />
      </View>
    );
  }

  return (
    <FlatList
      data={categories}
      keyExtractor={item => item.name}
      renderItem={({ item }) => (
        <TouchableOpacity onPress={() => loadVideos(item.file)} style={styles.cat}>
          <Text style={{ fontSize: 18 }}>{item.name}</Text>
          <Text style={{ color: '#888' }}>{item.count} videos</Text>
        </TouchableOpacity>
      )}
    />
  );
}

const styles = StyleSheet.create({
  cat: { padding: 20, borderBottomWidth: 1, borderColor: '#eee' },
  video: { flexDirection: 'row', padding: 10, borderBottomWidth: 1, borderColor: '#eee' },
  thumb: { width: 120, height: 68, marginRight: 10, borderRadius: 5 }
});
